Practical 5: BFS (8 Puzzle)

from collections import deque
 
goal = [1, 2, 3, 4, 5, 6, 7, 8, 0]
moves = {"UP": -3, "DOWN": 3, "LEFT": -1, "RIGHT": 1}
 
def print_state(state):
    for i in range(0, 9, 3):
        print(state[i:i+3])
    print()
 
def get_neighbors(state):
    neighbors = []
    zero_index = state.index(0)
    for move, pos in moves.items():
        new_index = zero_index + pos
        if move == "LEFT" and zero_index % 3 == 0: continue
        if move == "RIGHT" and zero_index % 3 == 2: continue
        if 0 <= new_index < 9:
            new_state = state[:]
            new_state[zero_index], new_state[new_index] = new_state[new_index], new_state[zero_index]
            neighbors.append(new_state)
    return neighbors
 
def bfs(start):
    queue = deque([(start, [])])
    visited = []
    while queue:
        state, path = queue.popleft()
        if state in visited: continue
        visited.append(state)
        print("Visiting State:")
        print_state(state)
        if state == goal:
            print("Goal Reached!")
            return path + [state]
        for neighbor in get_neighbors(state):
            queue.append((neighbor, path + [state]))
    return None
 
start = [1, 2, 3, 4, 0, 6, 7, 5, 8]
solution = bfs(start)
print("Solution Steps:")
for step in solution:
    print_state(step)

Practical 6: A* Search

import heapq
 
goal = [1, 2, 3, 4, 5, 6, 7, 8, 0]
 
def print_state(state):
    for i in range(0, 9, 3):
        print(state[i:i+3])
    print()
 
def manhattan(state):
    distance = 0
    for i in range(9):
        if state[i] == 0: continue
        goal_index = goal.index(state[i])
        x1, y1 = divmod(i, 3)
        x2, y2 = divmod(goal_index, 3)
        distance += abs(x1 - x2) + abs(y1 - y2)
    return distance
 
def get_neighbors(state):
    neighbors = []
    zero = state.index(0)
    moves = [-3, 3, -1, 1]
    for move in moves:
        new = zero + move
        if move == -1 and zero % 3 == 0: continue
        if move == 1 and zero % 3 == 2: continue
        if 0 <= new < 9:
            new_state = state[:]
            new_state[zero], new_state[new] = new_state[new], new_state[zero]
            neighbors.append(new_state)
    return neighbors
 
def a_star(start):
    heap = []
    heapq.heappush(heap, (manhattan(start), 0, start, []))
    visited = []
    while heap:
        f, g, state, path = heapq.heappop(heap)
        if state in visited: continue
        visited.append(state)
        print(f"Visiting (f={f}, g={g}):")
        print_state(state)
        if state == goal:
            print("Goal Reached!")
            return path + [state]
        for neighbor in get_neighbors(state):
            heapq.heappush(heap, (
                g + 1 + manhattan(neighbor),
                g + 1,
                neighbor,
                path + [state]
            ))
    return None
 
start = [1, 2, 3, 4, 0, 6, 7, 5, 8]
solution = a_star(start)
print("Solution Steps:")
for step in solution:
    print_state(step)

Practical 7: Minimax

import math
 
board = [" " for _ in range(9)]
 
def print_board():
    print()
    for i in range(0, 9, 3):
        print(board[i], "|", board[i+1], "|", board[i+2])
    print()
 
def check_winner(b):
    win_positions = [
        [0,1,2],[3,4,5],[6,7,8],
        [0,3,6],[1,4,7],[2,5,8],
        [0,4,8],[2,4,6]
    ]
    for pos in win_positions:
        if b[pos[0]] == b[pos[1]] == b[pos[2]] != " ":
            return b[pos[0]]
    if " " not in b:
        return "Draw"
    return None
 
def minimax(b, depth, is_maximizing):
    result = check_winner(b)
    if result == "O": return 1
    elif result == "X": return -1
    elif result == "Draw": return 0
 
    if is_maximizing:
        best = -math.inf
        for i in range(9):
            if b[i] == " ":
                b[i] = "O"
                score = minimax(b, depth+1, False)
                b[i] = " "
                best = max(best, score)
        return best
    else:
        best = math.inf
        for i in range(9):
            if b[i] == " ":
                b[i] = "X"
                score = minimax(b, depth+1, True)
                b[i] = " "
                best = min(best, score)
        return best
 
def ai_move():
    best_score = -math.inf
    move = -1
    for i in range(9):
        if board[i] == " ":
            board[i] = "O"
            score = minimax(board, 0, False)
            board[i] = " "
            if score > best_score:
                best_score = score
                move = i
    board[move] = "O"
    print(f"AI chooses position {move}")
 
def play_game():
    print("Positions are from 0 to 8")
    print_board()
    while True:
        pos = int(input("Enter your move (0-8): "))
        if board[pos] != " ":
            print("Invalid move!")
            continue
        board[pos] = "X"
        print("\nYour Move:")
        print_board()
        if check_winner(board): break
        ai_move()
        print("\nAI Move:")
        print_board()
        if check_winner(board): break
    result = check_winner(board)
    print("Game Over:", result)
 
play_game()

Practical 8: Genetic Algorithm

import random
 
# 1. Initial Population
population = [random.randint(0, 31) for _ in range(4)]
 
def fitness(x):
    return x ** 2   # Maximize x^2
 
for generation in range(5):
    # 2. Evaluate Fitness
    population = sorted(population, key=fitness, reverse=True)
 
    # 3. Selection (top 2)
    parents = population[:2]
 
    # 4. Crossover + Mutation
    child1 = (parents[0] + parents[1]) // 2
    child2 = parents[0] - 1   # Simple mutation
 
    # 5. New Generation
    population = parents + [child1, child2]
 
print("Best solution found by GA:", population[0])
print("Fitness value:", fitness(population[0]))

Practical 9: Alpha-Beta

import math
 
def alpha_beta(depth, node_index, is_max, scores, target_depth, alpha, beta):
    if depth == target_depth:
        return scores[node_index]
 
    if is_max:
        best = -math.inf
        for i in range(2):
            val = alpha_beta(depth+1, node_index*2+i, False, scores, target_depth, alpha, beta)
            best = max(best, val)
            alpha = max(alpha, best)
            if beta <= alpha:   # Prune
                break
        return best
    else:
        best = math.inf
        for i in range(2):
            val = alpha_beta(depth+1, node_index*2+i, True, scores, target_depth, alpha, beta)
            best = min(best, val)
            beta = min(beta, best)
            if beta <= alpha:   # Prune
                break
        return best
 
scores = [3, 5, 2, 9, 12, 5, 23, 23]
target_depth = int(math.log(len(scores), 2))
 
optimal_value = alpha_beta(0, 0, True, scores, target_depth, -math.inf, math.inf)
print("The optimal value with Alpha-Beta Pruning is:", optimal_value)
